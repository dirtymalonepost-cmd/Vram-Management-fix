set UID_NOW (id -u)

sudo systemctl disable --now dmemcg-appslice-limit.service 2>/dev/null
sudo rm -f /etc/systemd/system/dmemcg-appslice-limit.service /usr/local/sbin/set-dmem-appslice-limit

printf '%s\n' \
'#!/bin/sh' \
'RESERVE_MIB=50    # Change this to 75, 100, etc.' \
'ID="$1"' \
'APP="/sys/fs/cgroup/user.slice/user-$ID.slice/user@$ID.service/app.slice"' \
'' \
'# Wait for DMEM to appear on app.slice.' \
'for i in $(seq 1 300); do' \
'    [ -r "$APP/dmem.max" ] && [ -r "$APP/dmem.current" ] && [ -r /sys/fs/cgroup/dmem.capacity ] && break' \
'    sleep 1' \
'done' \
'' \
'[ -r "$APP/dmem.max" ] && [ -r "$APP/dmem.current" ] || exit 1' \
'' \
'set -- $(grep -m1 /vram /sys/fs/cgroup/dmem.capacity)' \
'DEVICE=$1' \
'CAPACITY=$2' \
'[ -n "$DEVICE" ] && [ -n "$CAPACITY" ] || exit 1' \
'' \
'TARGET=$((CAPACITY - RESERVE_MIB * 1024 * 1024))' \
'set -- $(grep -m1 /vram "$APP/dmem.current")' \
'CURRENT=$2' \
'[ -z "$CURRENT" ] || [ "$CURRENT" -le "$TARGET" ] || exit 1' \
'' \
'printf "%s %s\n" "$DEVICE" "$TARGET" > "$APP/dmem.max" || exit 1' \
'echo "VRAM capacity: $CAPACITY bytes"' \
'echo "Safety margin: $RESERVE_MIB MiB"' \
'echo "app.slice limit: $TARGET bytes"' \
| sudo tee /usr/local/sbin/set-dmem-appslice-limit >/dev/null

sudo chmod 755 /usr/local/sbin/set-dmem-appslice-limit

printf '%s\n' \
'[Unit]' \
'Description=Apply app.slice VRAM safety limit' \
'After=user@__UID__.service graphical.target dmemcg-booster-system.service' \
'Wants=user@__UID__.service dmemcg-booster-system.service' \
'' \
'[Service]' \
'Type=oneshot' \
'ExecStart=/usr/local/sbin/set-dmem-appslice-limit __UID__' \
'RemainAfterExit=yes' \
'TimeoutStartSec=300' \
'' \
'[Install]' \
'WantedBy=graphical.target' \
| string replace -a __UID__ $UID_NOW \
| sudo tee /etc/systemd/system/dmemcg-appslice-limit.service >/dev/null

sudo systemctl daemon-reload
sudo systemctl enable dmemcg-appslice-limit.service
sudo systemctl restart dmemcg-appslice-limit.service
systemctl status dmemcg-appslice-limit.service --no-pager -l
